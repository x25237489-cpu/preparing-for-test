Configure either Mode A (Public Cloudflare Worker Proxy) where keys stay locked inside server-side secrets on GitHub Pages, or Mode B (RAM-Only Session Vault) for immediate browser testing with your Groq (gsk_...) or Gemini (AIza...) key without writing to disk or code.

Unified Cloudflare Worker Proxy (Weather + Groq / Gemini Secrets) 
export default {
  async fetch(request, env) {
    const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "Content-Type" };
    if (request.method === "OPTIONS") return new Response(null, { headers: cors });

    const url = new URL(request.url);
    // Route 1: Key-gated Weather API (Visual Crossing or OpenWeatherMap)
    if (url.pathname === "/weather") {
      const lat = url.searchParams.get("lat");
      const lon = url.searchParams.get("lon");
      const mode = url.searchParams.get("mode"); // "forecast" | "history"
      const provider = url.searchParams.get("provider") || "visualcrossing";
      let upstream = "";
      if (provider === "openweathermap") {
        upstream = `https://api.openweathermap.org/data/2.5/forecast?lat=${lat}&lon=${lon}&units=metric&appid=${env.WEATHER_API_KEY}`;
      } else if (mode === "history") {
        upstream = `https://weather.visualcrossing.com/VisualCrossingWebServices/rest/services/timeline/${lat},${lon}/2020-10-01/2026-03-31?unitGroup=metric&include=days&key=${env.WEATHER_API_KEY}`;
      } else {
        upstream = `https://weather.visualcrossing.com/VisualCrossingWebServices/rest/services/timeline/${lat},${lon}/next48hours?unitGroup=metric&include=hours&key=${env.WEATHER_API_KEY}`;
      }
      const wxRes = await fetch(upstream);
      return new Response(await wxRes.text(), { headers: { ...cors, "Content-Type": "application/json" } });
    }

    // Route 2: Key-gated AI Briefings (Groq Llama-3.3-70B or Gemini 3 Flash)
    if (url.pathname === "/ai") {
      const body = await request.json();
      const aiProvider = url.searchParams.get("provider") || (env.GROQ_API_KEY ? "groq" : "gemini");
      if (aiProvider === "groq") {
        const groqRes = await fetch("https://api.groq.com/openai/v1/chat/completions", {
          method: "POST",
          headers: { "Content-Type": "application/json", "Authorization": `Bearer ${env.GROQ_API_KEY}` },
          body: JSON.stringify(body)
        });
        return new Response(await groqRes.text(), { headers: { ...cors, "Content-Type": "application/json" } });
      }
      const aiRes = await fetch(
        `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${env.GEMINI_API_KEY}`,
        { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) }
      );
      return new Response(await aiRes.text(), { headers: { ...cors, "Content-Type": "application/json" } });
    }
    return new Response("Winter Watch Secure Proxy Active", { headers: cors });
  }
};
